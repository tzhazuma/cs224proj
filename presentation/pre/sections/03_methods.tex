\begin{frame}{Four prompting strategies}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{description}[L4]
        \item[L1] direct: code $\to$ \code{\{label, confidence\}}
        \item[L2] few-shot: + fixed dev examples (cache-friendly prefix)
        \item[L3] analysis-guided: + Clang SA warnings for this unit
        \item[L4] neuro-symbolic: LLM lists heap facts, \textbf{Datalog} decides
      \end{description}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item Same prompt text for DeepSeek and Qwen
        \item Temperature 0, thinking mode off
        \item Prompts frozen in git before any eval run
        \item L3 latency includes Clang; L4 includes Souffl\'e
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  {\small Decision models answer the same question as options (\emph{leak / no leak}) and we read the option
  probability; L4 does not apply to them.}
\end{frame}

\begin{frame}[fragile]{L4: the LLM extracts facts, Datalog derives the leak}
  \centering
  \begin{tikzpicture}[node distance=4mm and 5mm]
    \node[box] (code) {unit\\code};
    \node[hero, right=of code] (llm) {LLM\\{\scriptsize DeepSeek / Qwen}};
    \node[data, right=8mm of llm, font=\scriptsize, align=left] (facts) {%
      \texttt{alloc realloc free}\\
      \texttt{assign pass ret call}\\
      \texttt{global dead(line)}};
    \node[hero, right=of facts] (dl) {Souffl\'e\\{\scriptsize \texttt{leak.dl}}};
    \node[warn, right=of dl] (out) {\texttt{leak(o,\,why)}\\{\scriptsize non-empty $\Rightarrow$ leak}};
    \draw[arr] (code) -- (llm);
    \draw[arr] (llm) -- node[lbl, above] {JSON} (facts);
    \draw[arr] (facts) -- (dl);
    \draw[arr] (dl) -- (out);
  \end{tikzpicture}
  \vspace{3pt}
\begin{lstlisting}[style=dl]
reach("main").
reach(g)         :- reach(f), call(f, g, l), !dead(l).          // + pass, retval
new_obj(o, v, f) :- alloc(o, v, f, l), reach(f), !dead(l).      // + realloc
pts(s, v, o)     :- new_obj(o, v, f), scope(f, v, s).           // Andersen-style
pts(sd, d, o)    :- assign(d, x, f, l), reach(f), !dead(l), ..., pts(sx, x, o).
freed(o)         :- free(v, f, l), reach(f), !dead(l), ..., pts(s, v, o).
(*\textcolor{OIblue}{\bfseries leak(o, "never\_freed")}*)  :- new_obj(o, _, _), !freed(o).
(*\textcolor{OIblue}{\bfseries leak(o, "realloc\_fail")}*) :- realloc(n, v, v, f, l), ..., pts(s, v, o), o != n.
\end{lstlisting}
  {\scriptsize Abridged from \texttt{juliet\_full/llm/l4/leak.dl} (v3). Flow-insensitive; parameters alias arguments both ways;
  facts on dead lines are ignored; facts whose line does not hold the statement are dropped first.}
\end{frame}
