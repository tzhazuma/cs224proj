\begin{frame}{Strategy ladder L1--L4}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{description}[L4]
        \item[L1] direct: code $\to$ \code{\{label, confidence\}}
        \item[L2] few-shot: + fixed dev examples
        \item[L3] analysis-guided: + Clang SA warnings
        \item[L4] neuro-symbolic: the LLM lists heap facts, \textbf{Datalog} decides
      \end{description}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{itemize}\small
        \item Same prompt text for DeepSeek and Qwen; temperature 0, thinking off

        \item Decision models read P(leak) for L1--L3
      \end{itemize}
    \end{column}
