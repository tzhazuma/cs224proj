\begin{frame}[fragile]{DeadFree adds a \code{free} that never runs}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{bad} + dead free $\Rightarrow$ still \alert{leak}
\begin{lstlisting}[style=unit]
static const int sym_7 = 5;
data = (char *)malloc(100);
(*\hlc{if (sym\_7 != 5) \{ free(data); \}}*)
\end{lstlisting}
      \textbf{good}: real free, always-true guard
\begin{lstlisting}[style=unit]
(*\hlc{if (sym\_7 == 5) \{ free(data); \}}*)
\end{lstlisting}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Placebo}: same guard, no free
\begin{lstlisting}[style=unit]
if (sym_7 != 5) { printLine("..."); }
\end{lstlisting}
      \vspace{4pt}
      \begin{itemize}
        \item Labels do not change
        \item ``has a \code{free}'' says nothing
        \item Placebo isolates the dead free
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  {\scriptsize All 3,316 units compile (gcc, clang); labels re-checked with ASan/LSan.\par}
\end{frame}
